### 2. SkipEvent（跳跃事件）
**功能**: 检测跳跃动作（双脚离地和落地）

**核心逻辑**:
```cpp
enum SkipStatus { UP, DOWN, UNKNOWN };

bool IsEvent(Landmarks kps, BBox box) {
    // 1. 计算颈部和脚踝中点的移动距离
    float neck_move = current_neck_y - last_neck_y;
    float ankle_move = current_ankle_y - last_ankle_y;
    
    // 2. 判断状态
    if (neck_move < -neck_up_distance && 
        ankle_move < -mid_ankle_up_distance) {
        skipStatus = UP;  // 起跳
    }
    else if (neck_move > neck_down_distance && 
             ankle_move > mid_ankle_down_distance) {
        skipStatus = DOWN;  // 落地
    }
    
    // 3. 状态转换计数
    if (lastSkipStatus == UP && skipStatus == DOWN) {
        count++;  // 完整跳跃周期
        return true;
    }
    
    lastSkipStatus = skipStatus;
    return false;
}
```

**关键参数**:
```json
{
  "skip_distance_threshold": 30,    // 跳跃高度阈值（像素）
  "neck_up_distance": 20,           // 颈部上升距离
  "neck_down_distance": 20,         // 颈部下降距离
  "mid_ankle_up_distance": 30,     // 脚踝上升距离
  "mid_ankle_down_distance": 30,   // 脚踝下降距离
  "down_frame_count": 3             // 落地稳定帧数
}
```

### 3. PushUpEvent（俯卧撑事件）
**功能**: 检测俯卧撑动作

**状态定义**:
```cpp
enum PUSH_UP_STATUS {
    STATUS_MID_DOWN,   // 下降中
    STATUS_DOWN,       // 最低点
    STATUS_MID_UP,     // 上升中
    STATUS_UP,         // 最高点
    STATUS_UNKNOWN
};

enum SPORT_STATUS {
    SPORT_STATUS_DOWN,    // 运动状态：下
    SPORT_STATUS_UP,      // 运动状态：上
    SPORT_STATUS_UNKNOWN
};
```

**核心判断逻辑**:
```cpp
bool IsEvent(Landmarks kps, BBox box) {
    // 1. 计算关键角度
    float elbow_angle = CalculateAngle(shoulder, elbow, wrist);
    float shoulder_angle = CalculateAngle(elbow, shoulder, hip);
    float body_angle = CalculateAngle(shoulder, hip, ankle);
    
    // 2. 检查身体姿态
    bool is_body_horizontal = abs(body_angle - 180) < 15;  // 身体水平
    bool is_leg_straight = knee_angle > foot_ready_angle;  // 腿伸直
    
    if (!is_body_horizontal || !is_leg_straight) {
        leg_bend_count++;
        if (leg_bend_count > leg_bend_frame_threshold) {
            return false;  // 姿态不标准
        }
    }
    
    // 3. 判断上下状态
    if (elbow_angle < down_threshold) {
        down_status = STATUS_DOWN;
        sport_status = SPORT_STATUS_DOWN;
    }
    else if (elbow_angle > up_threshold) {
        up_status = STATUS_UP;
        sport_status = SPORT_STATUS_UP;
    }
    
    // 4. 状态转换计数
    if (sport_status == SPORT_STATUS_UP && 
        last_sport_status == SPORT_STATUS_DOWN) {
        count++;  // 完成一次俯卧撑
        
        // 评分
        if (leg_bend_count > 0) {
            score = count + un_reach_ratio;  // 不标准动作
        } else {
            score = count;  // 标准动作
        }
        
        reset_status();
        return true;
    }
    
    return false;
}
```

**关键参数**:
```json
{
  "arm_ready_angle": 150,           // 手臂准备角度
  "foot_ready_angle": 150,          // 腿部伸直角度
  "down_threshold": 90,             // 肘关节下降阈值
  "up_threshold": 160,              // 肘关节上升阈值
  "un_reach_down_threshold": 100,  // 不达标下降阈值
  "un_reach_up_threshold": 140,    // 不达标上升阈值
  "un_reach_ratio": 0.5,           // 不达标得分比例
  "leg_bend_frame_threshold": 2    // 腿弯曲容忍帧数
}
```

### 4. SquatEvent（深蹲事件）
**功能**: 检测深蹲动作

**核心逻辑**:
```cpp
bool IsEvent(Landmarks kps, BBox box) {
    // 1. 构建人体姿态
    HumanBodyPose currentPose(kps);
    
    // 2. 计算关键距离
    // 颈部到髋部中点的垂直距离
    double neck_to_hip = currentPose.GetNeck().y - 
                         currentPose.GetMidHip().y;
    
    // 参考姿态（站立）的颈部到髋部距离
    double refer_neck_to_hip = referBodyPose.GetNeck().y - 
                                referBodyPose.GetMidHip().y;
    
    // 3. 判断深蹲
    // 当前距离 < 参考距离 * squatScale 时判定为深蹲
    bool is_squat = neck_to_hip < (refer_neck_to_hip * squatScale);
    
    // 4. 检测身体移动（防止作弊）
    bool is_moving = neckMovingEvent->IsEvent(kps, box) || 
                     midHipMovingEvent->IsEvent(kps, box);
    
    if (is_moving) {
        leave_count_frame++;
        if (leave_count_frame > 5) {
            return false;  // 移动过多，不计数
        }
    }
    
    // 5. 状态转换
    if (is_squat && last_status == STAND) {
        // 下蹲
        last_status = SQUAT;
    }
    else if (!is_squat && last_status == SQUAT) {
        // 站起
        count++;
        last_status = STAND;
        leave_count_frame = 0;
        return true;
    }
    
    return false;
}
```

**关键参数**:
```json
{
  "squatScale": 0.4,              // 深蹲比例（颈部到髋部距离）
  "moving_threshold": 50,         // 移动阈值（像素）
  "leave_count_threshold": 5      // 移动容忍帧数
}
```

### 5. JumpingEvent（跳跃事件 - 垂直跳）
**功能**: 检测垂直跳远距离

**核心逻辑**:
```cpp
enum Status { UP, DOWN, UNKNOWN };

bool IsEvent(Landmarks kps, BBox box) {
    HumanBodyPose currentPose(kps);
    
    // 1. 获取关键点位置
    double neck_y = currentPose.GetNeck().y;
    double sub_joint_y = currentPose.GetJoint(subJoint).y;  // 默认脚踝
    
    // 2. 计算连续移动距离
    double neck_move = neck_y - previousNeckPositionY;
    double sub_move = sub_joint_y - previousSubJointPositionY;
    
    // 上升时累加距离
    if (neck_move < 0) {  // Y轴向上为负
        neckContinueUpWardDistance += abs(neck_move);
    } else {
        neckContinueUpWardDistance = 0;
    }
    
    if (sub_move < 0) {
        subJointContinueUpWardDistance += abs(sub_move);
    } else {
        subJointContinueUpWardDistance = 0;
    }
    
    // 3. 判断起跳
    if (neckContinueUpWardDistance > jumpingDistance && 
        subJointContinueUpWardDistance > jumpingDistance) {
        status = UP;
    }
    
    // 4. 判断落地
    if (status == UP && neck_move > 0 && sub_move > 0) {
        status = DOWN;
        
        // 检查有效期
        long current_time = GetCurrentTimeMs();
        if (validityDelay > 0) {
            if (current_time - lastCompletionDate < validityDelay) {
                // 还在有效期内
                return true;
            }
        }
        
        lastCompletionDate = current_time;
        count++;
        
        // 记录最大跳跃高度
        if (subShoulderMax > max_jump_height) {
            max_jump_height = subShoulderMax;
        }
        
        // 重置
        subShoulderMax = 0;
        status = UNKNOWN;
        return true;
    }
    
    // 5. 记录最大高度
    double shoulder_sub_distance = currentPose.GetShoulder().y - sub_joint_y;
    if (shoulder_sub_distance > subShoulderMax) {
        subShoulderMax = shoulder_sub_distance;
    }
    
    previousNeckPositionY = neck_y;
    previousSubJointPositionY = sub_joint_y;
    
    return false;
}
```

**关键参数**:
```json
{
  "jumpingDistance": 50,          // 跳跃判定距离（像素）
  "subJoint": "MidAnkle",         // 辅助关节（脚踝）
  "validityDelay": 1000,          // 有效期（毫秒）
  "onceEventFlag": true           // 是否只计数一次
}
```

### 6. ArmHangEvent（悬挂事件 - 引体向上）
**功能**: 检测引体向上动作

**核心逻辑**:
```cpp
bool IsEvent(Landmarks kps, BBox box) {
    // 1. 计算关键角度
    float left_elbow = CalculateAngle(left_shoulder, left_elbow, left_wrist);
    float right_elbow = CalculateAngle(right_shoulder, right_elbow, right_wrist);
    
    // 2. 计算下巴高度
    float nose_y = kps.points[0].y;
    float wrist_y = (left_wrist.y + right_wrist.y) / 2;
    float chin_height = wrist_y - nose_y;  // 下巴相对手腕的高度
    
    // 3. 判断拉起
    if (chin_height < chin_threshold && 
        left_elbow < arm_angle_up && 
        right_elbow < arm_angle_up) {
        status = PULL_UP;
    }
    
    // 4. 判断悬挂
    if (chin_height > 0 && 
        left_elbow > arm_angle_down && 
        right_elbow > arm_angle_down) {
        status = HANG_DOWN;
    }
    
    // 5. 状态转换计数
    if (status == HANG_DOWN && last_status == PULL_UP) {
        count++;
        return true;
    }
    
    last_status = status;
    return false;
}
```

**关键参数**:
```json
{
  "chin_threshold": -50,          // 下巴高度阈值（负值表示高于手腕）
  "arm_angle_up": 70,             // 拉起时肘关节角度
  "arm_angle_down": 160,          // 悬挂时肘关节角度
  "body_swing_threshold": 20      // 身体摆动容忍度
}
```

### 7. DribbleEvent（运球事件）
**功能**: 检测篮球/足球运球

**核心逻辑**:
```cpp
bool IsEvent(Landmarks kps, BBox box, BBox ball_box) {
    // 1. 计算手部中心
    Point hand_center;
    hand_center.x = (left_wrist.x + right_wrist.x) / 2;
    hand_center.y = (left_wrist.y + right_wrist.y) / 2;
    
    // 2. 计算球心
    Point ball_center;
    ball_center.x = (ball_box.x1 + ball_box.x2) / 2;
    ball_center.y = (ball_box.y1 + ball_box.y2) / 2;
    
    // 3. 计算距离
    float distance = EuclideanDistance(hand_center, ball_center);
    
    // 4. 判断球是否被控制
    bool is_controlled = distance < control_distance_threshold;
    
    // 5. 检测运球动作（球的上下运动）
    float ball_move_y = ball_center.y - last_ball_y;
    
    if (is_controlled) {
        if (ball_move_y > dribble_down_threshold) {
            ball_status = DOWN;  // 球下降
        }
        else if (ball_move_y < -dribble_up_threshold) {
            ball_status = UP;    // 球上升
        }
        
        // 状态转换计数
        if (ball_status == UP && last_ball_status == DOWN) {
            dribble_count++;
            return true;
        }
        
        last_ball_status = ball_status;
    }
    
    last_ball_y = ball_center.y;
    return false;
}
```

**关键参数**:
```json
{
  "control_distance_threshold": 150,  // 控制距离阈值
  "dribble_down_threshold": 30,      // 运球下降阈值
  "dribble_up_threshold": 30,        // 运球上升阈值
  "min_dribble_frequency": 0.5,      // 最小运球频率（Hz）
  "max_dribble_frequency": 3.0       // 最大运球频率（Hz）
}
```

```cpp
bool IsEvent(Landmarks kps, BBox box, BBox ball_box) {
    // 1. 计算脚到球的距离
    float left_ankle_dist = Distance(left_ankle, ball_center);
    float right_ankle_dist = Distance(right_ankle, ball_center);
    float min_dist = min(left_ankle_dist, right_ankle_dist);
    
    // 2. 判断触球
    bool is_touch = min_dist < touch_threshold;
    
    // 3. 检测移动
    float ball_move = Distance(ball_center, last_ball_center);
    bool is_ball_moving = ball_move > move_threshold;
    
    // 4. 计数逻辑
    if (is_touch && is_ball_moving) {
        step_count++;
        return true;
    }
    
    last_ball_center = ball_center;
    return false;
}
```

### 9. ParallelBarDipEvent（双杠屈臂伸事件）
**功能**: 检测双杠屈臂伸动作

**核心逻辑**:
```cpp
bool IsEvent(Landmarks kps, BBox box) {
    // 1. 计算双臂角度
    float left_arm = CalculateAngle(left_shoulder, left_elbow, left_wrist);
    float right_arm = CalculateAngle(right_shoulder, right_elbow, right_wrist);
    
    // 2. 计算身体垂直度
    float body_angle = CalculateAngle(shoulder, hip, ankle);
    bool is_vertical = abs(body_angle - 180) < 15;
    
    // 3. 判断屈臂
    if (left_arm < arm_bend_angle && right_arm < arm_bend_angle && is_vertical) {
        status = ARM_BEND;
    }
    
    // 4. 判断伸臂
    if (left_arm > arm_straight_angle && right_arm > arm_straight_angle) {
        status = ARM_STRAIGHT;
    }
    
    // 5. 状态转换
    if (status == ARM_STRAIGHT && last_status == ARM_BEND) {
        count++;
        return true;
    }
    
    last_status = status;
    return false;
}
```

```json
{
```

### 11. SerpentineRunningEvent（蛇形跑事件）
**功能**: 检测蛇形跑轨迹

**核心逻辑**:
```cpp
bool IsEvent(Landmarks kps, BBox box) {
    // 1. 记录身体中心轨迹
    Point center = GetBodyCenter(kps);
    trajectory.push_back(center);
    
    // 2. 检测方向变化
    if (trajectory.size() >= 3) {
        Vector dir1 = trajectory[n-1] - trajectory[n-2];
        Vector dir2 = trajectory[n] - trajectory[n-1];
        
        float angle = AngleBetween(dir1, dir2);
        
        if (abs(angle) > direction_change_threshold) {
            turn_count++;
        }
    }
    
    // 3. 判断完成
    if (turn_count >= required_turns && 
        trajectory.size() > min_trajectory_length) {
        return true;
    }
    
    return false;
}
```

**关键参数**:
```json
{
  "required_turns": 4,              // 需要的转向次数
  "direction_change_threshold": 30, // 方向变化阈值（度）
  "min_trajectory_length": 100      // 最小轨迹长度
}
```

- **配置目录**: `item_kps_configs/push_up/`

- **判定逻辑**:
  ```
  1. 检测身体水平（肩-髋-踝角度 ≈ 180°）

  3. 肘关节下降（< 90°）判定为"下"
  4. 肘关节上升（> 160°）判定为"上"

  6. 不标准动作打折计分
  ```

- **使用事件**: SquatEvent + MovingEvent + AngleEvent

### 16. 实心球投掷 (Solid Ball Throw / 实心球 SXQ)
- **配置目录**: `yolo_v8_pose_object_byte_track/`
- **使用模型**: 
  - **人体检测**: YOLOv8 Pose
  - **目标检测**: YOLOv8/YOLOv7 (实心球检测)
  - **跟踪**: ByteTrack (人体) + IOUTracker (实心球)
- **关键点**: 肩(5,6)、肘(7,8)、腕(9,10)、髋(11,12) + 实心球位置

#### 算法流程
```
1. 人体姿态检测
   - 检测投掷者的全身关键点
   - ByteTrack 跟踪人体

2. 实心球目标检测
   - YOLOv8检测实心球位置
   - IOUTracker 跟踪实心球轨迹

3. 投掷姿态识别
   - 准备姿态：手持球，手臂后摆
   - 投掷动作：手臂前挥
   - 出手时刻：球离开手部

4. 投掷距离测量
   - 起始位置：投掷线
   - 落地位置：球第一次接触地面的位置
   - 计算投掷距离

5. 动作评估
   - 投掷角度（约45度最佳）
   - 出手速度
   - 身体协调性
   - 是否犯规（踩线等）
```

#### 检测逻辑
```cpp
bool IsThrowEvent() {
    // 1. 检测手部与球的距离
    float hand_ball_dist = CalculateDistance(wrist, ball_center);
    
    // 2. 判断持球状态
    if (hand_ball_dist < hold_threshold) {
        is_holding = true;
    }
    
    // 3. 检测投掷动作（手臂挥动）
    float arm_angle = CalculateAngle(shoulder, elbow, wrist);
    float arm_velocity = CalculateVelocity(wrist_positions);
    
    // 准备阶段：手臂后摆
    if (wrist.x < shoulder.x && is_holding) {
        status = PREPARE;
    }
    
    // 投掷阶段：手臂前挥
    if (wrist.x > shoulder.x && arm_velocity > throw_velocity_threshold) {
        status = THROWING;
    }
    
    // 出手时刻：球离开手部
    if (hand_ball_dist > release_threshold && status == THROWING) {
        status = RELEASED;
        
        // 记录出手位置和速度
        release_position = ball_center;
        release_velocity = CalculateBallVelocity();
        
        return true;  // 投掷动作完成
    }
    
    return false;
}

// 计算投掷距离
float CalculateThrowDistance() {
    // 追踪球的轨迹
    TrackBallTrajectory(ball_positions);
    
    // 找到球第一次落地的位置
    Point landing_position = FindFirstLanding(ball_positions);
    
    // 计算距离
    float distance = EuclideanDistance(start_line, landing_position);
    
    return distance;
}
```

```cpp
// 1. 起跑检测
bool CheckStartLine(SportInfo* sportInfo, x3::FrameMessage& frameMessage) {
    // 检测人员是否在起跑线区域内
    cv::Point2f ankle_mid = GetMidAnkle(kps);  // 脚踝中点
    
    // 判断是否在起跑区域
    if (cv::pointPolygonTest(init_area, ankle_mid, false) > 0) {
        sportInfo->in_area_count++;
        
        // 连续在区域内超过阈值，标记为准备就绪
        if (sportInfo->in_area_count > min_track_length) {
            sportInfo->is_ready = true;
        }
    }
    
    return sportInfo->is_ready;
}

// 2. 冲线检测
bool CheckCrossLine(SportInfo* sportInfo, x3::FrameMessage& frameMessage) {
    cv::Point2f ankle_mid = GetMidAnkle(kps);
    
    // 判断脚踝中点与终点线的位置关系
    int position = PointPositionRelativeToLine(ankle_mid, end_line);
    
    // 记录当前位置
    if (position > 0) {
        sportInfo->line_front_count++;  // 线前计数
    } else {
        sportInfo->line_behind_count++;  // 线后计数
    }
    
    // 判断是否完成冲线：从线前到线后
    if (sportInfo->line_front_count >= min_line_front && 
        sportInfo->line_behind_count > 0) {
        
        // 记录冲线时间
        sportInfo->cross_time = GetCurrentTime();
        sportInfo->is_cross = true;
        
        // 计算成绩（秒）
        int64_t duration = (sportInfo->cross_time - start_time) / 1000;
        sportInfo->score = duration;
        
        return true;
    }
    
    return false;
}

// 3. 抢跑检测
bool CheckFalseStart(SportInfo* sportInfo) {
    // 在开始信号发出前离开起跑区域
    if (sportInfo->cross_time > 0 && 
        sportInfo->cross_time < official_start_time) {
        sportInfo->is_foul = true;
        sportInfo->foul_reason = "抢跑";
        return true;
    }
    return false;
}
```

```cpp
// 1. 圈数计算
void AdvanceRun(x3::FrameMessage& frameMessage) {
    for (auto& runner : runners) {
        // 获取脚踝位置
        cv::Point2f ankle_mid = GetMidAnkle(runner.kps);
        
        // 计算与起点线的距离
        float current_dist = CalculateDistanceToLine(ankle_mid, start_line);
        
        // 第一次记录距离
        if (runner.first_dist == 0) {
            runner.first_dist = current_dist;
            runner.first_dist_time = GetCurrentTime();
        }
        
        // 更新最后距离
        runner.last_dist = current_dist;
        runner.last_dist_time = GetCurrentTime();
        
        // 检测过线（距离从大变小）
        if (runner.first_dist > cycle_size/2 && current_dist < cycle_size/2) {
            runner.lap_count++;  // 圈数+1
            
            // 计算圈速
            int64_t lap_time = runner.last_dist_time - runner.first_dist_time;
            runner.lap_speeds.push_back(lap_time / 1000.0);
            
            // 重置第一次距离
            runner.first_dist = current_dist;
            runner.first_dist_time = runner.last_dist_time;
        }
    }
}

// 2. 完成时间预测
float PredictFinishTime(LongRunner& runner) {
    // 已完成的距离
    float completed_distance = runner.lap_count * cycle_size + runner.last_dist;
    
    // 计算平均速度（米/秒）
    float avg_speed = completed_distance / 
                     ((runner.last_dist_time - start_time) / 1000.0);
    
    // 剩余距离
    float remaining_distance = max_distance - completed_distance;
    
    // 预测完成时间
    float predict_time = remaining_distance / avg_speed;
    
    return predict_time;
}

// 3. 终点冲线检测
bool CheckFinish(LongRunner& runner) {
    // 计算完成的距离
    float completed_distance = runner.lap_count * cycle_size + runner.last_dist;
    
    // 判断是否达到目标距离
    if (completed_distance >= max_distance) {
        runner.is_cross = true;
        runner.cross_time = GetCurrentTime();
        
        // 计算总用时（秒）
        runner.total_time = (runner.cross_time - start_time) / 1000.0;
        
        // 计算平均配速（秒/100米）
        runner.avg_pace = runner.total_time / (max_distance / 100.0);
        
        return true;
    }
    
    return false;
}

// 4. 号码识别与确认
void RecNum(cv::Mat& mat, x3::FrameMessage& frameMessage) {
    for (auto& runner : runners) {
        // OCR识别号码
        std::string num = OCRRecognize(mat, runner.body_box);
        
        // 添加到候选列表
        runner.num_infos.push_back({num, 1});
        
        // 统计识别次数最多的号码
        std::map<std::string, int> num_count;
        for (auto& info : runner.num_infos) {
            num_count[info.race_num]++;
        }
        
        // 确认号码（超过阈值）
        for (auto& [num, count] : num_count) {
            if (count >= num_min_rec_time) {
                runner.raceNum = num;
                runner.is_confirm = true;
                break;
            }
        }
    }
}
```
